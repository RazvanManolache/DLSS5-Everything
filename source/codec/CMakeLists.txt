cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(phoenix-pyrowave LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(PYROWAVE_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../.build-deps/pyrowave" CACHE PATH "Pinned upstream Pyrowave checkout")
if(NOT EXISTS "${PYROWAVE_SOURCE}/Granite/third_party/volk/volk.c")
    message(FATAL_ERROR "Run scripts/Prepare-Pyrowave.ps1 before building.")
endif()
set(GRANITE_PLATFORM "null" CACHE STRING "" FORCE)
foreach(feature RENDERER TOOLS VULKAN_SYSTEM_HANDLES VULKAN_SPIRV_CROSS VULKAN_SHADER_MANAGER_RUNTIME_COMPILER VULKAN_FOSSILIZE
        SHADER_COMPILER_OPTIMIZE ANDROID_APK_FILESYSTEM RENDERDOC_CAPTURE)
    set(GRANITE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(GRANITE_POSITION_INDEPENDENT ON CACHE BOOL "" FORCE)
set(GRANITE_SHIPPING ON CACHE BOOL "" FORCE)
add_subdirectory("${PYROWAVE_SOURCE}/Granite" granite EXCLUDE_FROM_ALL)
set(PYROWAVE_BUILD_SOURCE "${PYROWAVE_SOURCE}")
add_subdirectory("${PYROWAVE_BUILD_SOURCE}" pyrowave EXCLUDE_FROM_ALL)
add_library(phoenix-pyrowave SHARED pyrowave-bridge.cpp "${PYROWAVE_BUILD_SOURCE}/pyrowave_c.cpp"
    "${PYROWAVE_SOURCE}/Granite/video/scaler.cpp")
target_include_directories(phoenix-pyrowave PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${PYROWAVE_BUILD_SOURCE}" "${PYROWAVE_BUILD_SOURCE}/shaders" "${PYROWAVE_SOURCE}/Granite/video")
target_link_libraries(phoenix-pyrowave PRIVATE pyrowave granite-vulkan granite-math granite-volk-headers)
target_compile_definitions(phoenix-pyrowave PRIVATE PYROWAVE_EXPORT_SYMBOLS PW_BUILD)
option(PW_DIAGNOSTICS "Enable upstream Vulkan initialization diagnostics for local testing" OFF)
if(PW_DIAGNOSTICS)
    target_compile_definitions(phoenix-pyrowave PRIVATE VULKAN_DEBUG)
endif()
add_executable(dlss5-pyrowave codec-worker.cpp)
target_link_libraries(dlss5-pyrowave PRIVATE phoenix-pyrowave)
